Future<void> _fonte() async {
  final loader = FontLoader('Nunito');
  for (final peso in ['Regular', 'Bold']) {
    final bytes = File('assets/fonts/Nunito-$peso.ttf').readAsBytesSync();
    loader.addFont(Future.value(ByteData.view(bytes.buffer)));
  }
  await loader.load();
}

Future<void> _salva(WidgetTester tester, String nome,
    {double ratio = 1.5, Key chave = _chave}) async {
  final b = tester.renderObject<RenderRepaintBoundary>(find.byKey(chave));
  await tester.runAsync(() async {
    final img = await b.toImage(pixelRatio: ratio);
    final data = await img.toByteData(format: ui.ImageByteFormat.png);
    img.dispose();
    Directory(_pasta).createSync(recursive: true);
    File('$_pasta/$nome.png').writeAsBytesSync(data!.buffer.asUint8List());
  });
}

void _tela(WidgetTester tester, Size s) {
  tester.view.physicalSize = s;
  tester.view.devicePixelRatio = 1;
  addTearDown(tester.view.resetPhysicalSize);
  addTearDown(tester.view.resetDevicePixelRatio);
}

Widget _grade(String titulo, List<String> cols, List<String> linhas,
    Widget Function(int l, int c) cel) {
  return Directionality(
    textDirection: TextDirection.ltr,
    child: MediaQuery(
      data: const MediaQueryData(disableAnimations: true),
      child: Align(
        alignment: Alignment.topLeft,
        child: RepaintBoundary(
          key: _chave,
          child: Container(
            color: Colors.white,
            padding: const EdgeInsets.all(6),
            child: Column(
              mainAxisSize: MainAxisSize.min,
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                Text(titulo, style: _txt.copyWith(fontSize: 15, fontWeight: FontWeight.bold)),
                Row(mainAxisSize: MainAxisSize.min, children: [
                  const SizedBox(width: 90),
                  for (final c in cols)
                    SizedBox(width: 260, child: Text(c, style: _txt, textAlign: TextAlign.center)),
                ]),
                for (var l = 0; l < linhas.length; l++)
                  Row(mainAxisSize: MainAxisSize.min, children: [
                    SizedBox(width: 90, child: Text(linhas[l], style: _txt)),
                    for (var c = 0; c < cols.length; c++) cel(l, c),
                  ]),
              ],
            ),
          ),
        ),
      ),
    ),
  );
}

/// Caixa (bbox) do que foi pintado, relativo à caixa do PetView.
Future<Rect?> _bbox(WidgetTester tester, Widget pet) async {
  // PetView 200x150 no centro de 600x500 transparente.
  await tester.pumpWidget(Directionality(
    textDirection: TextDirection.ltr,
    child: MediaQuery(
      data: const MediaQueryData(disableAnimations: true),
      child: Align(
        alignment: Alignment.topLeft,
        child: RepaintBoundary(
          key: _chave,
          child: SizedBox(
            width: 600,
            height: 500,
            child: Stack(children: [Positioned(left: 200, top: 175, child: pet)]),
          ),
        ),
      ),
    ),
  ));
  await tester.pump();
  final b = tester.renderObject<RenderRepaintBoundary>(find.byKey(_chave));
  late Uint8List px;
  await tester.runAsync(() async {
    final img = await b.toImage();
    px = (await img.toByteData(format: ui.ImageByteFormat.rawRgba))!.buffer.asUint8List();
    img.dispose();
  });
  int? x0, y0, x1, y1;
  for (var y = 0; y < 500; y++) {
    for (var x = 0; x < 600; x++) {
      if (px[(y * 600 + x) * 4 + 3] > 20) {
        x0 = x0 == null || x < x0 ? x : x0;
        x1 = x1 == null || x > x1 ? x : x1;
        y0 = y0 == null || y < y0 ? y : y0;
        y1 = y1 == null || y > y1 ? y : y1;
      }
    }
  }
  if (x0 == null) return null;
  return Rect.fromLTRB(x0 - 200.0, y0! - 175.0, x1! - 200.0 + 1, y1! - 175.0 + 1);
}

  testWidgets('transição de humor: quadros', (tester) async {
    _tela(tester, const Size(1900, 1300));
    final casos = [
      (Species.capybara, Mood.content, Mood.missingYou),
      (Species.owl, Mood.content, Mood.missingYou),
      (Species.fox, Mood.content, Mood.missingYou),
      (Species.penguin, Mood.missingYou, Mood.radiant),
      (Species.cat, Mood.sleepy, Mood.radiant),
    ];
    const tempos = [0, 50, 100, 175, 250, 320, 400];
    for (final reduzido in [false, true]) {
      final quadros = <List<ui.Image>>[];
      for (final (sp, de, para) in casos) {
        final linha = <ui.Image>[];
        Widget cena(Mood m) => Directionality(
              textDirection: TextDirection.ltr,
              child: MediaQuery(
                data: MediaQueryData(disableAnimations: reduzido),
                child: Align(
                  alignment: Alignment.topLeft,
                  child: RepaintBoundary(
                    key: _chave,
                    child: Container(
                      color: _bg,
                      width: 260,
                      height: 210,
                      alignment: Alignment.center,
                      child: PetView(species: sp, mood: m, activity: Activity.idle, coat: 0, interativo: false),
                    ),
                  ),
                ),
              ),
            );
        await tester.pumpWidget(cena(de));
        await tester.pump(const Duration(milliseconds: 500));
        await tester.pumpWidget(cena(para));
        var t0 = 0;
        for (final t in tempos) {
          await tester.pump(Duration(milliseconds: t - t0));
          t0 = t;
          final b = tester.renderObject<RenderRepaintBoundary>(find.byKey(_chave));
          late ui.Image img;
          await tester.runAsync(() async {
            img = await b.toImage(pixelRatio: 1);
          });
          linha.add(img);
          if (t == 175) {
            final n = find.descendant(of: find.byType(PetView), matching: find.byType(Opacity)).evaluate().length;
            File('$_pasta/medidas-transicao.txt').writeAsStringSync(
                'reduzido=$reduzido ${sp.name} ${de.name}->${para.name}: Opacity em 175ms = $n\n',
                mode: FileMode.append);
          }
        }
        quadros.add(linha);
        await tester.pumpWidget(const SizedBox());
        await tester.pump(const Duration(seconds: 30));
      }
      // Monta a folha.
      await tester.runAsync(() async {
        final rec = ui.PictureRecorder();
        final c = Canvas(rec);
        c.drawRect(const Rect.fromLTWH(0, 0, 7 * 262.0, 5 * 212.0), Paint()..color = Colors.white);
        for (var l = 0; l < quadros.length; l++) {
          for (var k = 0; k < quadros[l].length; k++) {
            c.drawImage(quadros[l][k], Offset(k * 262.0, l * 212.0), Paint());
          }
        }
        final img = await rec.endRecording().toImage(7 * 262, 5 * 212);
        final data = await img.toByteData(format: ui.ImageByteFormat.png);
        File('$_pasta/transicao${reduzido ? '-reduzido' : ''}.png')
            .writeAsBytesSync(data!.buffer.asUint8List());
      });
    }

void mainBrinca() {
  testWidgets('brincadeira da coruja: fases', (tester) async {
    _tela(tester, const Size(1900, 700));
    final imgs = <ui.Image>[];
    for (final (sp, a) in [(Species.owl, Activity.swim), (Species.owl, Activity.graze), (Species.cat, Activity.swim)]) {
      await tester.pumpWidget(Directionality(
        textDirection: TextDirection.ltr,
        child: MediaQuery(
          data: const MediaQueryData(),
          child: Align(
            alignment: Alignment.topLeft,
            child: RepaintBoundary(
              key: _chave,
              child: Container(
                color: _bg, width: 260, height: 210, alignment: Alignment.center,
                child: PetView(species: sp, mood: Mood.radiant, activity: a, coat: 0, interativo: false),
              ),
            ),
          ),
        ),
      ));
      for (var i = 0; i < 7; i++) {
        await tester.pump(const Duration(milliseconds: 180));
        final b = tester.renderObject<RenderRepaintBoundary>(find.byKey(_chave));
        await tester.runAsync(() async {
          imgs.add(await b.toImage(pixelRatio: 1));
        });
      }
      await tester.pumpWidget(const SizedBox());
      await tester.pump(const Duration(seconds: 30));
    }
    await tester.runAsync(() async {
      final rec = ui.PictureRecorder();
      final c = Canvas(rec);
      for (var i = 0; i < imgs.length; i++) {
        c.drawImage(imgs[i], Offset((i % 7) * 262.0, (i ~/ 7) * 212.0), Paint());
      }
      final img = await rec.endRecording().toImage(7 * 262, 3 * 212);
      final data = await img.toByteData(format: ui.ImageByteFormat.png);
      File('$_pasta/coruja-brincadeira.png').writeAsBytesSync(data!.buffer.asUint8List());
    });
  });
}
